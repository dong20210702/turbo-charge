#配置更改实时生效
#CYCLE_TIME必须大于0，其他配置必须大于等于0
#所有配置都必须为整数，且最大可设为2147483647，若超过此值，会出现BUG（具体原因自行百度）

#循环间隔时间（单位:s）
#默认为1秒，不建议大于5秒
#间隔时间越长，程序越省电，但越有可能导致抢充电协议，使得手机疯狂断充
CYCLE_TIME=1

#是否启用强制显示28℃功能（1为启用，其他数字为禁用）
#充电时把系统看到的电池温度固定为28℃，让系统自身的过温降流不介入
#真实温度超过下面的FORCE_TEMP_MAX时，程序会自动把真实温度交还给系统，不会一直骗下去
#不想承担任何温度风险就把本项设为0：充电会明显变慢，但这是系统原本的保护策略
FORCE_TEMP=1

#最大充电电流（单位:μA。经测试超过手机理论最大充电电流不会出现bug，所以保持默认即可）
#注意：小于100000的值会让内核直接停止充电，程序会自动按100000处理
CURRENT_MAX=50000000

#是否关闭阶梯式充电（1为关闭，其他数字为不关闭）
#想让“降速发生在高电量段”必须设为1
#设为0时系统会在电量过半后就自己往下阶梯降流，这时候模块把电流上限写成再大也没用，
#表现就是“峰值只能维持一小会儿然后一路掉到10W” —— 这是最常见的原因
#注意：某些电池老化的手机在关闭阶梯充电后，电量低于15%时充电会反复断充，所以低电量段仍会开启阶梯充电
STEP_CHARGING_DISABLED=1
#是否添加温控（1为添加，其他数字为不添加）
TEMP_CTRL=1
#是否添加电量控制（1为添加，其他数字为不添加）
POWER_CTRL=0

#此配置关闭阶梯式充电时生效
#关闭阶梯式充电的电量阈值（电量大于等于此值时关闭阶梯式充电，整数）
#阶梯式充电状态切换时有几率造成1-2次的断充
#注意，某些手机电池健康状态不好时关闭阶梯式充电后电量低于15%时充电会疯狂断充
STEP_CHARGING_DISABLED_THRESHOLD=15

#以下两个配置添加电量控制时生效
#停止充电的电量阈值（电量大于等于此值时停止充电）
CHARGE_STOP=95
#开始充电的电量阈值（电量小于等于此值时开始充电）
CHARGE_START=80

#以下三个配置添加温控时生效
#降低充电电流的温度阈值（单位:℃）
#这里用的是“手机温度传感器”（默认wifi_therm，主板/WiFi区），不是电池温度
#实测满速充电叠加重负载时它能到54℃，所以阈值设在52会被长期触发、把功率压在26W
#电池过温保护由下面的FORCE_TEMP_MAX单独负责，两者互不替代
TEMP_MAX=58
#当高于温度阈值时的最大充电电流（单位:μA）
#注意：小于100000的值会让内核直接停止充电，程序会自动按100000处理
#7A 约 29.5W，是实测在本机散热条件下能长时间站住的最高档位
#（8A 只能维持约 8.5 分钟，之后温度必然越限）
HIGHEST_TEMP_CURRENT=7000000
#恢复快充的温度阈值（单位:℃）
#与TEMP_MAX之间要留出足够间隔，否则会反复进出限流
#实测wifi_therm在限流后稳定在49~51℃，所以出阈值设在52基本等于出不来，设48会反复进出
RECHARGE_TEMP=50

#是否启用“伪”旁路供电功能（1为启用，其他数字为禁用）
#使用多线程+popen执行Shell命令来获取前台应用，所以开启后会消耗更多的系统资源
#为尽量减少系统资源的消耗，每5秒获取一次前台应用包名，不可更改
#未经完全测试，如果日志没有提示该功能失效且开启后无效，请及时前往Github进行反馈
#开启后请配置/data/adb/turbo-charge/bypass_charge.txt文件
BYPASS_CHARGE=0

#==================== 智能分段充电与温度安全阀（新增） ====================

#是否启用智能分段充电（1为启用，其他数字为关闭）
#启用后不再无脑把CURRENT_MAX写进内核，而是以PEAK_CURRENT为满速基准，按电量分档降流：
#  低于STAGE_START        100%
#  STAGE_START ~ 90%       70%
#  90% ~ 95%               45%
#  95% 以上                25%
#关掉则始终写CURRENT_MAX（旧版行为）
#需要说明：这里改的是“允许的电流上限”，真正能跑多大电流由充电IC、电池电压和温度决定，
#写更大的数值不会凭空变快；分段的意义是把降速点从系统默认的“电量过半”挪到80%以后
SMART_CHARGE=1

#分段起点（单位:%），低于此值给满速
#想让降速更晚就调大，最大99；调太小会让大电流区间变短、充得更慢
STAGE_START=80

#智能分段模式下的满速基准电流（单位:μA，8000000=8A）
#按4.4V电池电压估算，8A约等于35W；想要更猛可以调到10000000，但发热会明显增加
PEAK_CURRENT=8000000

#==================== 温度梯度（四级阶梯） ====================
#限流不再是一刀切的“熔断”，而是四级递降，每级各自带解除阈值。
#设计依据（v80 实测，9%→100%，全程亮屏）：
#  8A 满速把电池从 45.2℃ 推到 52.1℃ 用 8.5 分钟，升温 +0.84℃/分；
#  降到 6A 后 5 分钟就回落到 48.3℃，此后整段停在 46~49℃，降温 -0.21℃/分。
#  线性外推：热平衡电流约 6.4A（≈27W），7A 的平衡温度约 54℃（≈29.5W）。
#  也就是说，想让功率贴近 30W，必须允许电池长时间停在 53~55℃。

#1级：进入阈值（单位:℃）
#FORCE_TEMP把系统看到的电池温度伪造成28℃之后，系统自身的过温保护等于失效，
#所以程序会独立读取电池附近的温度传感器（thermal子系统里的mtktsbts*，本模块从不写入它）
#超过此值把电流降到 HIGHEST_TEMP_CURRENT（7A）
#注意：该传感器装在电池附近的主板上，读数比电芯本体偏高3~8℃，
#      但它跟随的是电池仓的温升，是目前唯一可信的过热判据
#电池充电的推荐温度上限是45℃，52℃已属激进档位，不建议继续往上调
FORCE_TEMP_MAX=52

#1级：解除阈值（单位:℃）
#跌破此值交回分段电流（0-80% 段是 8A）
#旧值 46℃ 是 v80 最大的浪费点：温度 07:10 就回到 48.3℃，却要等到 46℃ 才放开，
#结果从 33% 到 82% 整整 24 分钟锁在 6A，白丢约 2.5Wh。
#49℃ 让温度一离开 52℃ 的峰值带就交回 8A，与 2级阈值(55℃)之间留足间隔
GUARD_RECOVER=49

#熔断时是否把真实温度交还给系统（0=不交还，1=交还）
#默认0：本机实测，系统一旦看到46℃以上的真实电池温度，会对充电器做开/关硬切断，
#      59分钟里电量从38%只涨到48%就不再增长，比限流慢充还差得多。
#      所以默认由模块自己限流降温，真实温度只记进日志的tbat_raw列供人工核对。
#设为1则恢复“交还真实温度”的老行为：系统的过温保护会真正生效，
#      但在这台机型上等于放弃快充，只在把电池当牺牲品不可接受时才需要打开
GUARD_HANDOVER=0

#2级：进入阈值（单位:℃）
#温度继续上升时从 7A 再降一档。7A 的平衡温度约 54℃，已经贴近此阈值，
#它平时不该被触发；真正的用途是逆光、车内、太阳直射这类环境，
#让温度缓慢爬过 55℃ 时有人接手，而不是一路顶到 3级
GUARD2_TEMP=55
#2级：电流上限（单位:μA），6000000=6A，约25W
GUARD2_CURRENT=6000000
#2级：解除阈值（单位:℃）
#跌破 52℃ 回到 1级（7A）。与 1级 52进/49出 交叉，形成 52~55℃ 的稳定带：
#温度落在这个区间时停在 1级，既不会整段放开到 8A，也不会掉回 6A
GUARD2_RECOVER=52

#临界温度（单位:℃）与对应的电流上限（单位:μA）
#超过CRITICAL_TEMP后，在熔断限流的基础上再降一档，避免长时间停在高温区
#4500000=4.5A，在4.2V电池上约19W，是实测能把温度往下拉的档位
#实测提示：这台机器亮屏重负载时SoC自身就能把电池仓加热到55℃以上，
#          此时继续8A充电等于叠加发热，是温度失控的主因
#v78实测：这一档设在60℃时等于没有——6A限流下电池真实温度峰值只到59.1℃，
#          全程一次都没触发过。前移到57℃才真正参与控制
CRITICAL_TEMP=57
CRITICAL_CURRENT=4500000

#临界降流的解除温度（单位:℃）
#3级：解除温度（单位:℃）
#不能沿用 1级的 GUARD_RECOVER(49℃)：一旦 57℃ 触发 3级，要等电池跌回 49℃ 才回到 2级，
#而充电中后段温度长期在 50~58℃ 之间，等于触发一次就永久锁在 4.5A 直到充满。
#54℃ 保留 3℃ 滞回，既能退出，又不会在 57℃ 附近反复横跳
CRITICAL_RECOVER=54

#临界降流的最短保持时间（单位:秒，0=不限制）
#v79实测：临界只保持了3.5分钟（57.0℃→55.4℃）就放回6A，随后立刻回升到56.3℃。
#          若那时没进恒压区，就会变成 6A→57℃→4.5A→53℃→6A 的周期性横跳，
#          平均功率反而比稳定在4.5A更低。180秒足够让温度真正脱离临界带
#3级和4级都受本项约束（离开高温区之前必须先保持满这个时长）
CRITICAL_HOLD=180

#==================== 4级限流（重负载兜底）====================
#亮屏打游戏/录屏这类场景，发热源是SoC而不是充电本身，降流压不住温度，
#但3级(4.5A)是原来的最后一级，到59℃左右模块就无档可退
#v84实测（11-22早，2%→97%）：全程CPU均76.4℃/峰97℃，电池温度与CPU相关系数0.788、
#      与充电电流相关系数-0.301。8A只跑9.4分钟，之后29.7分钟停在4.5A，
#      电池均温56.9℃、峰值59.5℃，43%的电量是在这个温度下灌进去的
#60℃是常见电芯的JEITA快充上限，长时间停留会实打实损耗循环寿命
#进档温度（单位:℃），必须比CRITICAL_TEMP至少高2℃，否则会被自动修正
CRITICAL2_TEMP=60
#4级的电流上限（单位:μA），2000000=2A，约8W。只求把温度拉回来
CRITICAL2_CURRENT=2000000
#4级解除温度（单位:℃），跌破后回3级(4.5A)
#与进档温度拉开3℃，配合CRITICAL_HOLD避免在60℃附近来回跳
CRITICAL2_RECOVER=58

#==================== 高温软启动 ====================
#插线瞬间电池已经偏热时，先用低电流把温度拉下来，再交给熔断/分段策略
#v79实测结论：这项没有收益，默认关闭（=0）
#      插线瞬间电池电流本身从1.2A自然爬升，4A限流前20秒完全不起作用；
#      等电流碰到4A时温度已经52.7℃，限流真正生效只有65秒、降到48.2℃就退出；
#      随后6A用16分钟又顶回57℃。净效果只是延后90秒，
#      还白扔了一段4A低功率（15.6W vs 25W），少充约0.5个点
#想重新启用就把它改成45~50
SOFT_START_TEMP=0
#软启动期间的电流上限（单位:μA），4000000=4A，约17W
SOFT_START_CURRENT=4000000
#温度降到该值以下提前结束软启动（单位:℃）
SOFT_START_RECOVER=48
#软启动最长持续时间（单位:秒）
#亮屏重负载时4A也可能降不到48℃，没有这个兜底会一直卡在4A，把充电时间白白拉长
SOFT_START_TIMEOUT=360

#==================== 输入限流提升（实验，默认关闭）====================
#本项 >0 时把全部 input_current_limit 节点单独设成该值（单位:μA），
#      不跟随电池电流，试图触发内核重新协商更高的电压档。建议先试 6000000（6A）
#风险：若充电器/线材不支持更高档位，写大电流会被 AICL 压回；
#      最坏情况是掉到 5V 档变成慢充。所以默认 0 关闭。
#      开启后请在充电开始1分钟内观察日志的 icl_a 与 pbat_w：
#      一旦发现功率反而下降，立刻改回 0
#注意：早期版本曾根据 icl_a 恒为 3.225A 反推出"适配器在 9V 档、输入上限 29W，
#      输出到不了 26W"，这个结论已被实测推翻——同一配置下输出达到过 34.5W。
#      icl_a 是静态寄存器读数，不能用来反推真实输入功率。本机也不暴露 VBUS 电压，
#      所以无法直接验证协商档位，只能靠输出功率变化间接判断
INPUT_CURRENT_BOOST=0

#==================== 诊断日志 ====================
#充电采样日志固定写入 /data/adb/turbo-charge/charge_log.csv，不需要配置
#充电中每5秒一行、未充电每30秒一行，超过4MB自动轮转为charge_log.old.csv
#导出方法（电脑上执行）：
#  adb pull /data/adb/turbo-charge/charge_log.csv
#各列含义见文件第一行表头，重点看：
#  tsensor_c  手机温度传感器读数（wifi_therm，本模块不写它）
#  tbat_raw   真实电池温度（thermal子系统mtktsbts*，本模块不写它）—— 防过热的唯一依据
#  tbat_c     电池温度节点（已伪装成28.0，仅用于确认伪装是否生效）
#  vusb_v/ibus_max_a  充电器实际协商到的电压和电流上限（判断67W有没有被识别）
#  ccc_max_a/ccc_a    内核实际接受的恒流上限与实际恒流值（判断写入有没有被系统改回）
#  applied_a/stage    本模块本轮写入的电流和所处策略档位
