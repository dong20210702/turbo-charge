#配置更改实时生效
#CYCLE_TIME必须大于0，其他配置必须大于等于0
#所有配置都必须为整数，且最大可设为2147483647，若超过此值，会出现BUG（具体原因自行百度）

#循环间隔时间（单位:s）
#默认为1秒，不建议大于5秒
#间隔时间越长，程序越省电，但越有可能导致抢充电协议，使得手机疯狂断充
CYCLE_TIME=1

#是否启用强制显示28℃功能（1为启用，其他数字为禁用）
#充电时把系统看到的电池温度固定为28℃，让系统自身的过温降流不介入
#真实温度超过下面的FORCE_TEMP_MAX时，程序会自动把真实温度交还给系统，不会一直骗下去
#不想承担任何温度风险就把本项设为0：充电会明显变慢，但这是系统原本的保护策略
FORCE_TEMP=1

#最大充电电流（单位:μA。经测试超过手机理论最大充电电流不会出现bug，所以保持默认即可）
#注意：小于100000的值会让内核直接停止充电，程序会自动按100000处理
CURRENT_MAX=50000000

#是否关闭阶梯式充电（1为关闭，其他数字为不关闭）
#想让“降速发生在高电量段”必须设为1
#设为0时系统会在电量过半后就自己往下阶梯降流，这时候模块把电流上限写成再大也没用，
#表现就是“峰值只能维持一小会儿然后一路掉到10W” —— 这是最常见的原因
#注意：某些电池老化的手机在关闭阶梯充电后，电量低于15%时充电会反复断充，所以低电量段仍会开启阶梯充电
STEP_CHARGING_DISABLED=1
#是否添加温控（1为添加，其他数字为不添加）
TEMP_CTRL=1
#是否添加电量控制（1为添加，其他数字为不添加）
POWER_CTRL=0

#此配置关闭阶梯式充电时生效
#关闭阶梯式充电的电量阈值（电量大于等于此值时关闭阶梯式充电，整数）
#阶梯式充电状态切换时有几率造成1-2次的断充
#注意，某些手机电池健康状态不好时关闭阶梯式充电后电量低于15%时充电会疯狂断充
STEP_CHARGING_DISABLED_THRESHOLD=15

#以下两个配置添加电量控制时生效
#停止充电的电量阈值（电量大于等于此值时停止充电）
CHARGE_STOP=95
#开始充电的电量阈值（电量小于等于此值时开始充电）
CHARGE_START=80

#以下三个配置添加温控时生效
#降低充电电流的温度阈值（单位:℃）
#这里用的是“手机温度传感器”（默认wifi_therm，主板/WiFi区），不是电池温度
#实测满速充电叠加重负载时它能到54℃，所以阈值设在52会被长期触发、把功率压在26W
#电池过温保护由下面的FORCE_TEMP_MAX单独负责，两者互不替代
TEMP_MAX=58
#当高于温度阈值时的最大充电电流（单位:μA）
#注意：小于100000的值会让内核直接停止充电，程序会自动按100000处理
#2A在4.2V电池上只有8.4W，一旦进入限流区间充电会慢得几乎停滞；
#4A约16.8W，实测仍会把34W打到16.6W，掉速太明显；
#6A约26W，熔断时只是从34W退到26W，用户基本感觉不到掉速，同时仍有降温作用
HIGHEST_TEMP_CURRENT=6000000
#恢复快充的温度阈值（单位:℃）
#与TEMP_MAX之间要留出足够间隔，否则会反复进出限流
#实测wifi_therm在限流后稳定在49~51℃，所以出阈值设在52基本等于出不来，设48会反复进出
RECHARGE_TEMP=50

#是否启用“伪”旁路供电功能（1为启用，其他数字为禁用）
#使用多线程+popen执行Shell命令来获取前台应用，所以开启后会消耗更多的系统资源
#为尽量减少系统资源的消耗，每5秒获取一次前台应用包名，不可更改
#未经完全测试，如果日志没有提示该功能失效且开启后无效，请及时前往Github进行反馈
#开启后请配置/data/adb/turbo-charge/bypass_charge.txt文件
BYPASS_CHARGE=0

#==================== 智能分段充电与温度安全阀（新增） ====================

#是否启用智能分段充电（1为启用，其他数字为关闭）
#启用后不再无脑把CURRENT_MAX写进内核，而是以PEAK_CURRENT为满速基准，按电量分档降流：
#  低于STAGE_START        100%
#  STAGE_START ~ 90%       70%
#  90% ~ 95%               45%
#  95% 以上                25%
#关掉则始终写CURRENT_MAX（旧版行为）
#需要说明：这里改的是“允许的电流上限”，真正能跑多大电流由充电IC、电池电压和温度决定，
#写更大的数值不会凭空变快；分段的意义是把降速点从系统默认的“电量过半”挪到80%以后
SMART_CHARGE=1

#分段起点（单位:%），低于此值给满速
#想让降速更晚就调大，最大99；调太小会让大电流区间变短、充得更慢
STAGE_START=80

#智能分段模式下的满速基准电流（单位:μA，8000000=8A）
#按4.4V电池电压估算，8A约等于35W；想要更猛可以调到10000000，但发热会明显增加
PEAK_CURRENT=8000000

#真实温度熔断阈值（单位:℃）
#FORCE_TEMP把系统看到的电池温度伪造成28℃之后，系统自身的过温保护等于失效，
#所以程序会独立读取电池自身的温度传感器（thermal子系统里的mtktsbts*，本模块从不写入它）
#超过此值立即把充电电流限制为HIGHEST_TEMP_CURRENT
#电池充电的推荐温度上限是45℃，49℃已属于激进档位，不建议继续往上调
#注意：这里判断的是“电池温度”，不是wifi_therm那种主板温度，两者实测能差5~7℃
#      日志 tbat_raw 列就是这个电池温度，务必对照这一列确认电池没有真的过热
FORCE_TEMP_MAX=49

#熔断解除阈值（单位:℃）
#必须低于上面的FORCE_TEMP_MAX，两者之间的区间是“保持上一次状态”
#相等或更高等于没有滞回：温度在阈值上下抖1℃，模块就会一秒放开8A、一秒收回2A，
#平均功率比一直熔断还低，而且反复冲击充电IC
GUARD_RECOVER=46

#熔断时是否把真实温度交还给系统（0=不交还，1=交还）
#默认0：本机实测，系统一旦看到46℃以上的真实电池温度，会对充电器做开/关硬切断，
#      59分钟里电量从38%只涨到48%就不再增长，比限流慢充还差得多。
#      所以默认由模块自己限流降温，真实温度只记进日志的tbat_raw列供人工核对。
#设为1则恢复“交还真实温度”的老行为：系统的过温保护会真正生效，
#      但在这台机型上等于放弃快充，只在把电池当牺牲品不可接受时才需要打开
GUARD_HANDOVER=0

#==================== 诊断日志 ====================
#充电采样日志固定写入 /data/adb/turbo-charge/charge_log.csv，不需要配置
#充电中每5秒一行、未充电每30秒一行，超过4MB自动轮转为charge_log.old.csv
#导出方法（电脑上执行）：
#  adb pull /data/adb/turbo-charge/charge_log.csv

#  tbat_c     电池温度节点（已伪装成28.0，仅用于确认伪装是否生效）
#  vusb_v/ibus_max_a  充电器实际协商到的电压和电流上限（判断67W有没有被识别）
#  ccc_max_a/ccc_a    内核实际接受的恒流上限与实际恒流值（判断写入有没有被系统改回）
#  applied_a/stage    本模块本轮写入的电流和所处策略档位
